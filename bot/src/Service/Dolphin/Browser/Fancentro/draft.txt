  async getDialogMessagesLive(username) {
    if (this.dialogMessagesEmitter) {
      await this.dialogMessagesEmitter.changeUser(username);
      return this.dialogMessagesEmitter;
    }

    // Open an empty page and import cookies.
    const page = await this.browser.newPage();
    try {
      const cookies = await this.profile.exportCookies();
      await page.setCookie(...cookies);
    } catch (error) {
      console.error('Error importing cookies:', error);
    }

    // Open devtools to handle WebSockets in the future.
    const devtoolsSession = await page.target().createCDPSession();
    await devtoolsSession.send('Network.enable');

    const wsSessions = [];

    const wsAuthCredentials = await (new Promise((resolve) => {
      let wsPrimaryUrl;

      const webSocketConnectionListener = ({url, requestId}) => {
        if (url.match(/^wss:\/\/im\.fancentro\.com/)) {
          wsPrimaryUrl = url;
        }
        wsSessions.push(requestId);
      }

      const outgoingWebSocketListener = ({response}) => {
        if (response.payloadData.match(/42\/fc,\["authentication",/)) {
          const payloadData = JSON.parse(response.payloadData.replace(/^42\/fc,/, ''));
          resolve({
            url: wsPrimaryUrl,
            key: payloadData[1].authKey,
            sessionHash: payloadData[1].sessionHash
          });

          devtoolsSession.off('Network.webSocketCreated', webSocketConnectionListener);
          devtoolsSession.off('Network.webSocketFrameSent', outgoingWebSocketListener);
        }
      }

      devtoolsSession.on('Network.webSocketCreated', webSocketConnectionListener);
      devtoolsSession.on('Network.webSocketFrameSent', outgoingWebSocketListener);

      page.goto(`https://fancentro.com/admin/messages`);
    }));

    for (const session of wsSessions) {
      await devtoolsSession.send('Network.closeConnection', {
        requestId: session
      });
    }

    // Start WS connection.
    await page.evaluate(async (wsAuthCredentials) => {
      return new Promise((resolve, reject) => {
        resolve(new WebSocket(wsAuthCredentials.url));
      });
    }, wsAuthCredentials);

    const messagesEmitter = new MessagesEmitter(devtoolsSession);
    await messagesEmitter.changeUser(username);

    return this.dialogMessagesEmitter = messagesEmitter;
  }
